============================================================
  GDB CHEAT SHEET
============================================================

КОМПИЛЯЦИЯ (обязательно с -g)
  gcc -g -o program program.c

ЗАПУСК
  gdb ./program                  запустить программу в gdb
  gdb ./program core             открыть core dump
  quit  / q                      выйти из gdb

------------------------------------------------------------
ЗАПУСК И УПРАВЛЕНИЕ ВЫПОЛНЕНИЕМ
------------------------------------------------------------
  run  / r                       запустить программу
  run arg1 arg2                  запустить с аргументами
  kill                           остановить выполнение

  continue  / c                  продолжить до следующего breakpoint
  next      / n                  следующая строка (не заходит в функцию)
  step      / s                  следующая строка (заходит в функцию)
  finish                         выполнить до конца текущей функции
  until 42                       выполнить до строки 42

------------------------------------------------------------
BREAKPOINTS
------------------------------------------------------------
  break main         / b main    поставить breakpoint на функцию
  break 25           / b 25      поставить breakpoint на строку 25
  break file.c:25                breakpoint в другом файле
  info breakpoints   / i b       список всех breakpoints
  delete 1           / d 1       удалить breakpoint №1
  delete                         удалить все breakpoints
  disable 1                      выключить breakpoint №1 (не удалять)
  enable 1                       включить обратно

WATCHPOINTS (срабатывают при изменении переменной)
  watch x                        остановить когда x изменится
  watch *ptr                     следить за значением по указателю

------------------------------------------------------------
ПРОСМОТР ДАННЫХ
------------------------------------------------------------
  print x            / p x       напечатать значение переменной x
  print *ptr                     напечатать значение по указателю
  print arr[0]                   элемент массива
  print &x                       адрес переменной

  print/x x                      напечатать в hex
  print/d x                      напечатать в decimal
  print/b x                      напечатать в binary
  print/c x                      напечатать как символ

  display x                      печатать x после каждого шага
  undisplay 1                    убрать display №1

  info locals                    все локальные переменные
  info args                      аргументы текущей функции

------------------------------------------------------------
ПАМЯТЬ
------------------------------------------------------------
  x/4xw &arr         examine: 4 слова (word=4байт) в hex начиная с &arr
  x/8xb ptr                      8 байт в hex
  x/s ptr                        строка (читать до \0)
  x/i $rip                       инструкция по текущему указателю

  Форматы x: x=hex, d=decimal, c=char, s=string, i=инструкция
  Размеры:   b=byte(1), h=half(2), w=word(4), g=giant(8)

------------------------------------------------------------
СТЕК ВЫЗОВОВ
------------------------------------------------------------
  backtrace  / bt                показать стек вызовов
  backtrace 5                    показать последние 5 фреймов
  frame 2    / f 2               переключиться на фрейм №2
  up                             подняться на один фрейм вверх
  down                           опуститься на один фрейм вниз

------------------------------------------------------------
ИСХОДНЫЙ КОД
------------------------------------------------------------
  list       / l                 показать текущую строку ±5
  list 20                        показать строку 20 ±5
  list main                      показать функцию main
  list 10,30                     показать строки 10-30

------------------------------------------------------------
РАБОТА С ТИПАМИ И СТРУКТУРАМИ
------------------------------------------------------------
  ptype x                        показать тип переменной x
  ptype struct item              показать поля структуры

  print *node                    напечатать всю структуру
  print node->data               поле структуры через указатель
  print node->next               следующий элемент списка

------------------------------------------------------------
УДОБНЫЕ НАСТРОЙКИ (добавить в ~/.gdbinit)
------------------------------------------------------------
  set print pretty on            красивый вывод структур
  set print array on             читаемый вывод массивов
  set pagination off             не останавливать вывод на --More--

------------------------------------------------------------
ТИПИЧНЫЙ СЕАНС ОТЛАДКИ
------------------------------------------------------------
  $ gcc -g -o list list.c
  $ gdb ./list
  (gdb) break main
  (gdb) run
  (gdb) next
  (gdb) print first
  (gdb) print *first
  (gdb) continue
  (gdb) quit
